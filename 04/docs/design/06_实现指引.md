## 4. 交互原语伪码（FUNC 编号连续，01 线可直译）

```gdscript
# ============ FUNC-001 Inspect 状态机 ============
# 输入：桌面(鼠标) / VR(单柄或双柄)已由适配层合成为原语参数
enum InspectState { IDLE, DRAG_ROTATE, PINCH_SCALE, TWO_HAND }

signal inspect_updated(delta_rot: Vector3, delta_zoom: float, target: Node3D)
signal inspect_bounds_hit(limit: String)           # 达限额时反馈（音效+UI 提示）

const ROTATE_SPEED_DESKTOP : float = 180.0        # 度/秒（全幅拖动）
const ZOOM_RANGE : Vector2 = Vector2(0.5, 3.0)    # FR-B1 限额
const PITCH_LIMIT : float = 30.0                  # 展厅级俯仰限 [TBD]

func inspect_start():
    if current_device.state == DeviceState.RUNNING and hit_inside_crate():
        emit_signal("inspect_bounds_hit", "running_lock")   # FR-B4 禁穿模
        return false
    state = DRAG_ROTATE
    return true

func inspect_drag(mouse_delta: Vector2) -> void:
    if state != DRAG_ROTATE: return
    # 公式：旋转角 = 屏幕位移 × rad/px 系数；竖直轴不限额
    var rot := Vector3(-mouse_delta.y, -mouse_delta.x, 0) * rad_per_px
    if abs(current_pitch + rot.x) > PITCH_LIMIT:
        # 越限处理：夹紧并提示（FR-B1 边界反馈）
        emit_signal("inspect_bounds_hit", "pitch")
        rot.x = clamp_pitch(current_pitch, rot.x)
    target.rotate_by(rot)
    emit_signal("inspect_updated", rot, 0.0, target)

func inspect_zoom(wheel_delta: float) -> void:
    # 公式：缩放速率 = 当前倍数 × 0.1 / 滚轮格；指数步进手感
    var next := clampf(current_zoom * (1.0 + 0.1 * wheel_delta),
                       ZOOM_RANGE.x, ZOOM_RANGE.y)
    if next == current_zoom:
        emit_signal("inspect_bounds_hit", "zoom")
        return
    apply_zoom(next, pivot = pointer_hit_point)    # 以指针热点为缩放中心
    emit_signal("inspect_updated", Vector3.ZERO, next - current_zoom, target)

# VR 双手（FR-B3）：两手位姿差值 → 旋转/缩放/平移
func vr_two_hand_update(left: XRPose, right: XRPose) -> void:
    state = TWO_HAND
    var prev_span: float = last_span
    var span: float = left.position.distance_to(right.position)
    # 公式：缩放 = 跨度比；旋转 = 两手中点方向角差；平移 = 中点位移
    apply_zoom(clampf(current_zoom * span / prev_span, ZOOM_RANGE.x, ZOOM_RANGE.y))
    target.rotate_by(Vector3(0, angle_delta(mid_dir), 0))
    target.position += mid_delta
    last_span = span
    emit_signal("inspect_updated", Vector3(0, angle_delta(mid_dir), 0),
                 span / prev_span, target)
```

```gdscript
# ============ FUNC-002 Grab 状态机 ============
signal grab_picked(obj: Node3D)
signal grab_released(obj: Node3D, result: String)  # "snapped" | "dropped"
signal grab_snap_anim_started(obj: Node3D, dur: float)

const SNAP_DIST : float = 0.15      # 回装判定距离阈值（米）FR-D2
const SNAP_ANGLE : float = 30.0     # 姿态偏差阈值（度）
const SNAP_ANIM : float = 0.4       # 卡合动画时长（秒）
const PICK_SCALE : float = 1.05     # 拾取视觉确认放大

func try_pick(hit: Node3D) -> bool:
    if hold_object != null: return false                 # 一次一件
    if not hit in allowed_pickable_set(): return false   # 非可抓集合
    if animation_busy(hit): return false                 # 动画占用
    hold_object = hit
    hit.scale *= PICK_SCALE                              # 视觉确认
    emit_signal("grab_picked", hit)
    return true

func move_hold(delta: Vector3) -> void:
    if hold_object == null: return
    # 阻尼：小件灵敏、整机级迟缓（重量感近似，阻尼系数查表）
    hold_object.position += delta * damping_factor(hold_object.mass_class)

func release() -> void:
    if hold_object == null: return
    var obj := hold_object
    hold_object = null
    obj.scale /= PICK_SCALE
    if meets_snap(obj):                                  # 回装判定
        emit_signal("grab_snap_anim_started", obj, SNAP_ANIM)
        tween_snap_back(obj, SNAP_ANIM)                  # 平移+旋转插值卡合
        play_sfx("snap_in")                              # 04 章音效 id
        emit_signal("grab_released", obj, "snapped")
    else:
        obj.state = LOOSE                                # 成为散件
        play_sfx("drop_settle")
        emit_signal("grab_released", obj, "dropped")

func is_meets_snap(obj: Node3D) -> bool:
    var d: float = obj.position.distance_to(obj.home_transform.origin)
    var a: float = obj.rotation_angle_to(obj.home_transform.basis)
    return d <= SNAP_DIST and a <= SNAP_ANGLE and not blocked_by_others(obj)
```

```gdscript
# ============ FUNC-003 Point 状态机 ============
signal point_selected(target)
signal point_cancelled(reason: String)

const UI_HOTZONE_PX : float = 24.0        # 桌面逻辑热区（最小）
const VR_HOTZONE_M : float = 0.04         # VR 3 m 处热区（米）
const PRESS_MOVE_THRESHOLD_PX : float = 5.0   # 防误触移动阈值
const VR_PREss_MOVE_THRESHOLD_M : float = 0.02

func on_pointer_pressed(at: Vector2) -> void:
    press_pos = at
    pressed_target = pick(at)              # UI 优先，其次 3D 目标
    if pressed_target != null:
        set_highlight(pressed_target, SELECTED)   # FR-B2：描边+虚线脉动

func on_pointer_moved(at: Vector2) -> void:
    if press_pos.distance_to(at) > PRESS_MOVE_THRESHOLD_PX:
        # ≥阈值：取消点击语义，转交 Grab/Inspect（01 章 §2.3 Fine rule）
        press_pos = null
        if pressed_target is Pickable3D: handoff_to_grab(pressed_target, at)
        else: handoff_to_inspect(at)

func on_pointer_released(at: Vector2) -> void:
    if press_pos == null: return          # 已被移动阈值取消
    var t = pressed_target
    clear_highlight(t)
    if t != null and dragging_was_not_moved():
        emit_signal("point_selected", t)   # 单击成立
        play_sfx("confirm")
    press_pos = null
```

enum DeviceState { COMPLETE, OPENED, DISASSEMBLING, RUNNING }
signal device_state_changed(from_state, to_state)

# 运行就绪正交子态
enum ReadyState { NOT_READY, READY, CRASHED }

var device_state := DeviceState.COMPLETE
var ready_state := ReadyState.NOT_READY
var transition_lock := false        # 过渡锁（01 章 §4.7）

func enter_COMPLETE():
    clear_loose_parts()             # 无散件
    run_entry.enabled = (ready_state == ReadyState.READY)   # 就绪亮/未就绪灰
func exit_COMPLETE(): pass
func input_COMPLETE(primitive, target):
    match primitive:
        Primitive.POINT:
            if target == run_entry:
                if ready_state != ReadyState.READY:
                    open_wizard()                    # FR-E3 引导向导
                    play_sfx("hint")
                else: request_open_power()           # →RUNNING
        Primitive.GRAB:
            if target == lid_handle: open_lid()      # 开盖动作
func open_lid():
    if transition_lock: return
    transition_lock = true
    play_sfx("lid_open")
    tween(lid, 0.8, closed_pose -> open_pose,
          on_finish = func():
              device_state = DeviceState.OPENED
              transition_lock = false
              emit_signal("device_state_changed", COMPLETE, OPENED))

func enter_OPENED(): guide_card.switch_tab("internal")   # 导览卡切页
func input_OPENED(primitive, target):
    match primitive:
        Primitive.GRAB:
            if target == lid: grab_lid_light()       # 上盖可轻移（[TBD] 待拍板）
            elif target.is_pickable_part():
                detach_first_part(target)            # 拆首件 →DISASSEMBLING
func detach_first_part(target):
    loose_set.append(target)
    device_state = DeviceState.DISASSEMBLING
    show_disassembly_panel()                        # FR-D3
    emit_signal("device_state_changed", OPENED, DISASSEMBLING)

func enter_DISASSEMBLING():
    run_entry.enabled = false                       # 拆解中禁开机（FR-D/FR-E）
    run_entry.show_lock_icon()
func input_DISASSEMBLING(primitive, target):
    match primitive:
        Primitive.POINT:
            if target == run_entry:
                toast("请先复原再开机"); play_sfx("deny")   # 禁止动作提示
            if target == restore_btn: restore_all()        # →COMPLETE
        Primitive.GRAB:
            if target.is_part(): try_pick(target)          # 拆/装

func restore_all():                                  # 一键复原 FR-D3d
    if transition_lock: return
    transition_lock = true
    for part in loose_set.sorted_by_tree_depth():    # 叶→根、由远及近
        tween(part, 0.15, part.home_pose, on_each = play_sfx("snap_in"))
    tween(lid, 0.5, lid.closed_pose, on_finish = func():
        loose_set.clear(); transition_lock = false
        device_state = DeviceState.COMPLETE
        emit_signal("device_state_changed", DISASSEMBLING, COMPLETE))

func enter_RUNNING():
    hdd_show_hud()                                   # FR-E6a
    screen.enable_output(runtime.output_sinks)       # CRT 大屏 FR-I2
    play_sfx("power_on"); start_device_ambience()
func input_RUNNING(primitive, target):
    match primitive:
        Primitive.GRAB: deny_grab("运行中不可拆解")   # FR-B4/E（拒止反馈）
        Primitive.POINT:
            if target == snapshot_btn: take_snapshot()      # FR-E6b
            if target == quit_btn: request_quit_confirm()   # FR-E7 二次确认
func exit_RUNNING(): stop_device_ambience(); hdd_hide_hud()
