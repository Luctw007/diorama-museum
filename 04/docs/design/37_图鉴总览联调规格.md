### 3.1 轮播状态机（ASCII 全文）

```
          ┌────────────┐  8 s Timer 到期      ┌──────────────┐  hover/focus/注视1.2s
          │ REC_IDLE   │ ──────────────────▶ │ REC_SHUFFLING│ ───────────────┐
          └────────────┘                     └──────┬───────┘                │
               ▲ 渲染完成                           │ 重摇≤3 后仍同组→提示    ▼
               │  [换一组]/跨日/参观权重更新         │ 成功换组          ┌─────────────┐
               └───────────────────────────────────┴────────────────▶│ REC_FROZEN   │
                                                                      └──────┬──────┘
                                                                             │ 移出悬停/焦点离开
                                                                             ▼
                                                                      （回 REC_IDLE，Timer 重置）
```

**伪代码（GDScript 风格，全文）**：

```gdscript
var st := REC_IDLE
var round_no := 0
var retries := 0

func _on_recommend_timer_timeout() -> void:
    if st == REC_FROZEN: return                     # 悬停暂停不重置计时
    shuffle_to_next()

func shuffle_to_next() -> void:
    st = REC_SHUFFLING
    round_no += 1
    var group := Recommender.pick_group(today(), round_no, profile)   # §3.2
    if group.id_set == last_group.id_set and retries < 3:
        retries += 1
        shuffle_to_next()                                              # 规则 3 重摇
        return
    retries = 0
    last_group = group
    CardRow.fade_render(group, 0.3)                                    # 26 号 §3.1 同款淡入
    if group.exhausted: ReasonText.show_notice("已看完全部推荐")
    st = REC_IDLE

func _on_card_hover(entered: bool) -> void:
    st = REC_FROZEN if entered else REC_IDLE        # 注视端由 Area3D 替代
    if entered: RecommendTimer.stop()
    else: RecommendTimer.start(8.0)

func _on_next_group_pressed() -> void:
    sfx.play("sfx_page"); shuffle_to_next()
```

### 3.2 推荐算法状态机（冷启动 → 参观后权重更新，伪代码全文）

```gdscript
class_name PediaRecommender
# S(i) = 0.40*IN + 0.35*PR + 0.25*RAND        (35 号 §2.1)

static func pick_group(day: String, round: int, profile: Dictionary) -> Dictionary:
    var rows := CatalogData.all_light_rows()          # 只取 id/status/device_id/reason/name/maker/year
    var done_pool := rows.filter(func(r): r.status == "done" and r.device_id != null)
    var rest_pool := rows.filter(func(r): not (r.status == "done" and r.device_id != null))
    var cold := profile.is_empty() or not profile.has("visits")
    var scores := {}
    for r in rows:
        var IN := minf(1.0, r.reason_len() / 50.0)
        IN += 0.25 if r.status == "done" else (0.15 if r.status == "researching" else 0.0)
        IN += minf(0.15, r.booking_count * 0.01)
        IN = minf(IN, 1.0)
        var PR := progress_of(r.id, profile, cold)    # 冷启动恒 1.0；参观/收藏查 35 号 2.1 表
        var RAND := hash01(day + round + r.id.left(8))# FNV-1a → [0,1)
        scores[r.id] = 0.40*IN + 0.35*PR + 0.25*RAND
    var top := rows.sorted_custom(scores.get).slice(0, 3)
    if not top.any(func(r): r in done_pool):          # done 保底（35 号 §2.2 规则 2）
        var best_done := done_pool.sorted_custom(scores.get).front()
        top[2] = best_done                           # 顶替分数最低卡
    return {"cards": top, "exhausted": rows.size() < 3}
```

**权重更新事件（状态迁移表）**：

| 事件 | 触发 | PR 变化 | 生效时机 |
|---|---|---|---|
| 冷启动读档 | profile 空/无 visits | 全量 PR=1.0 | 立即 |
| 参观前（详情停留 ≥2 s / 进展厅） | 详情页计时器 | 该 id PR→0.6（首访）/0.3（≥2 次） | 下轮换组后（当前三卡会话内不动） |
| 收藏 | [收藏] 点击 | 该 id PR→0.15 | 同上下轮生效 |
| 跨日 | last_daily_seed ≠ today | seed 全重摇 + round_no 归零 | 进首页时先于渲染 |
